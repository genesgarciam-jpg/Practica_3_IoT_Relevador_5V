#include "BluetoothSerial.h"

// Verificación de que el Bluetooth esté activo
#if !defined(CONFIG_BT_ENABLED) || !defined(CONFIG_BLUEDROID_ENABLED)
#error Bluetooth no esta habilitado. Habilitalo en el menu de la tarjeta.
#endif

BluetoothSerial SerialBT;
const int RELAY_PIN = 26;

// Función para activar el relé (Manda 0V / Tierra)
void encenderRele() {
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW);
}

// Función para desactivar el relé (Alta impedancia / Pin flotante)
void apagarRele() {
  pinMode(RELAY_PIN, INPUT);
}

void setup() {
  Serial.begin(115200);
  
  // Estado inicial de seguridad: Apagado
  apagarRele();

  // Nombre Bluetooth que aparecerá en tu celular
  SerialBT.begin("ESP32_Relevador_Horeb"); 
  
  Serial.println("==================================================");
  Serial.println("Bluetooth iniciado con exito.");
  Serial.println("Dispositivo listo: 'ESP32_Relevador_Horeb'");
  Serial.println("==================================================");
}

void loop() {
  // Verificar si se recibió algún dato por Bluetooth desde la App
  if (SerialBT.available()) {
    String comando = SerialBT.readStringUntil('\n');
    comando.trim();         // Limpia espacios y saltos de línea
    comando.toLowerCase();  // Convierte el texto a minúsculas

    if (comando == "on") {
      encenderRele();
      SerialBT.println(">> Bombilla ENCENDIDA");
      Serial.println("Comando Bluetooth recibido: 'on' -> Bombilla ENCENDIDA");
    } 
    else if (comando == "off") {
      apagarRele();
      SerialBT.println(">> Bombilla APAGADA");
      Serial.println("Comando Bluetooth recibido: 'off' -> Bombilla APAGADA");
    }
  }
}
